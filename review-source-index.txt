function roundCompletion(m){
  let total=0,complete=0,missingFields=0;
  document.querySelectorAll('#sheet-'+m+' .card').forEach(card=>{
    if(priorMastered(m,Number(card.dataset.n),round(m)))return;
    const fields=Array.from(card.querySelectorAll('.blank-field,.response'));
    total++;
    if(!fields.length){complete++;return;}
    const missing=fields.filter(field=>!String(field.value||'').trim()).length;
    if(missing===0)complete++;
    missingFields+=missing;
  });
  return {total,complete,missingQuestions:total-complete,missingFields,allComplete:(total===0||complete===total)};
}
function roundEvaluated(m,r){
  let evaluated=true;
  document.querySelectorAll('#sheet-'+m+' .card').forEach(card=>{
    if(priorMastered(m,Number(card.dataset.n),r))return;
    const key=id(m,Number(card.dataset.n),r);
    if(!state.tick[key]&&!state.graded[key])evaluated=false;
  });
  return evaluated;
}
